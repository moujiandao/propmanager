## SQL to Run (if not already done)

```sql
-- landlord_profiles (if starting fresh)
create table public.landlord_profiles (
  id uuid primary key references auth.users(id) on delete cascade,
  name text not null,
  email text not null,
  created_at timestamptz default now()
);
alter table public.landlord_profiles enable row level security;
create policy "Landlords can read own profile" on public.landlord_profiles for select using (auth.uid() = id);
create policy "Landlords can update own profile" on public.landlord_profiles for update using (auth.uid() = id);

-- monthly_rent on tenant_profiles
alter table public.tenant_profiles add column if not exists monthly_rent numeric;
```
